# 赛题四(停止标志)过弯卡死修复:加大 l_weight,让路径贴参考线、不切角。
# 默认 l_weight=1.0 太小(相比 ddl_weight=1000/dddl_weight=50000 可忽略),
# 过急弯(半径约 7.5m)时路径优化器会切角走内圈,把车带到 -0.73(越过道路边界
# -0.7145),导致 lane_follow_path 的路径边界判断失败、QP 不可行、车卡死。
# 提高到 100 后,"保持居中"和"减小曲率"的代价相当,路径贴住车道中心,
# 车不会再漂出边界。只对本场景 INTERSECTION_CRUISE 阶段生效,不影响全局。
path_optimizer_config {
  l_weight: 100.0
}
